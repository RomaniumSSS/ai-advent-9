const $=id=>document.getElementById(id);
let current=null,viewVersion=0,state={active_request:null},submissionVersion=null,changingVersion=null,stateSequence=0,listSequence=0;
const kinds={assignment:'Условие',discussion:'Обсуждение',personal_note:'Личная заметка'};
const terminal=s=>!['running','submitted'].includes(s);
const seconds=x=>typeof x==='number'?`${x.toFixed(2)} с`:'не измерено';
function node(tag,text,className){const e=document.createElement(tag);if(text!==undefined)e.textContent=text;if(className)e.className=className;return e}
function token(){return {cid:current,version:viewVersion}}
function valid(t){return current===t.cid&&viewVersion===t.version}
function remember(cid){try{localStorage.setItem('day29-full-chat-conversation',cid)}catch{}}
async function api(path,body){const response=await fetch(path,body===undefined?{}:{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(body)});const value=await response.json();if(!response.ok)throw Error(value.error||`HTTP ${response.status}`);return value}
function controls(){const active=state.active_request,changing=changingVersion===viewVersion,sending=submissionVersion===viewVersion;$('send').disabled=changing||sending||!!active||!current;$('new').disabled=changing;$('reset').disabled=changing||!current;$('conversations').disabled=changing;$('status').textContent=changing?'Меняю разговор…':active?(active.conversation_id===current?'Локальный запрос выполняется…':'Завершается запрос другого разговора'):sending?'Отправка…':'Готов к вопросу'}
function showError(error,t){if(!valid(t))return;$('error').textContent=String(error);if(changingVersion===t.version)changingVersion=null;if(submissionVersion===t.version)submissionVersion=null;controls()}
function beginChange(cid){viewVersion++;current=cid;changingVersion=viewVersion;submissionVersion=null;$('error').textContent='';$('question').value='';render({turns:[]});controls();return token()}
async function refreshState(t=token()){const sequence=++stateSequence;const value=await api('/api/state');if(!valid(t)||sequence!==stateSequence)return false;state=value;$('scope').textContent=`Все ${value.index.chunk_count} фрагмента · ${value.index.document_count} документа · ${value.model}`;$('identity').textContent=`Scope: all\nИндекс: ${value.index.generation}\nSHA: ${value.index.db_sha256}\nTemperature ${value.options.temperature} · ctx ${value.options.num_ctx} · output ${value.options.num_predict} · think=false`;controls();return true}
function sourceView(source){const box=node('details',undefined,'source');box.dataset.requestId=source.request_id;box.append(node('summary',`${source.source_ref} · ${kinds[source.source]||source.source} · ${source.document_id} · ${source.chunk_id}`));for(const part of source.excerpts||[]){const p=node('div',undefined,'part');const author=part.attribution==='ambiguous'?'Атрибуция неоднозначна':part.author||'Автор неизвестен';p.append(node('div',`${author} · ${part.date||'Дата неизвестна'} · позиции ${part.start_char??'?'}–${part.end_char??'?'}`,'attribution'),node('pre',part.text));box.append(p)}const whole=node('details',undefined,'whole');whole.append(node('summary','Оригинальный фрагмент целиком'),node('pre',source.text,'source-full'));box.append(whole);return box}
function render(conversation){$('messages').replaceChildren();if(!conversation.turns.length){$('messages').append(node('p','Задайте новый вопрос. История этого разговора пока пуста.','empty'));return}for(const turn of conversation.turns){const box=node('article',undefined,'turn');box.dataset.requestId=turn.request_id;box.dataset.state=turn.status;if(turn.status==='error'||turn.status==='interrupted')box.classList.add('error');box.append(node('p',turn.question,'question'));let answer=turn.answer?.answer||'';if(!terminal(turn.status))answer='Ищу источники и выполняю локальную генерацию…';else if(turn.status==='interrupted')answer='Запрос был прерван перезапуском. Автоматического повтора не было.';else if(turn.error)answer=`${turn.error.phase}: ${turn.error.message}`;box.append(node('p',answer,'answer'));if(turn.query?.intent==='followup')box.append(node('p',`Продолжение темы: ${turn.query.anchor}`,'meta'));box.append(node('p',`request ${turn.request_id} · ${turn.status} · генерация ${seconds(turn.times?.generation)} · весь запрос ${seconds(turn.times?.total)}`,'meta'));if(turn.citations_valid)box.append(node('p','Ссылки проверены по свежему контексту. Смысловая поддержка автоматически не оценивалась.','support'));for(const source of turn.answer?.sources||[])box.append(sourceView(source));$('messages').append(box)}}
async function list(t=token()){const sequence=++listSequence;const result=await api('/api/conversations');if(!valid(t)||sequence!==listSequence)return null;$('conversations').replaceChildren();for(const c of result.conversations){const option=node('option',`Разговор ${c.conversation_id.slice(0,6)} · ${c.turn_count} реплик`);option.value=c.conversation_id;$('conversations').append(option)}if(current)$('conversations').value=current;return result.conversations}
async function hydrate(conversation,t){if(!valid(t))return;if(conversation.conversation_id!==t.cid)throw Error('conversation_id_mismatch');render(conversation);changingVersion=null;controls();await list(t);if(!valid(t))return;await refreshState(t);if(!valid(t))return;const pending=conversation.turns.find(turn=>!terminal(turn.status));if(pending)void poll(pending.request_id,t)}
async function open(cid){const t=beginChange(cid);try{const conversation=await api(`/api/conversations/${cid}`);if(!valid(t))return;remember(cid);await hydrate(conversation,t)}catch(error){showError(error,t)}}
async function newChat(){const t=beginChange(null);try{const conversation=await api('/api/conversations',{});if(!valid(t))return;current=conversation.conversation_id;t.cid=current;remember(current);await hydrate(conversation,t)}catch(error){showError(error,t)}}
async function reset(){if(!current)return;const cid=current,t=beginChange(cid);try{const conversation=await api('/api/reset',{conversation_id:cid});if(!valid(t))return;await hydrate(conversation,t)}catch(error){showError(error,t)}}
async function poll(rid,t){try{for(;;){await new Promise(resolve=>setTimeout(resolve,750));if(!valid(t))return;const job=await api(`/api/requests/${rid}`);if(!valid(t))return;const conversation=await api(`/api/conversations/${t.cid}`);if(!valid(t))return;render(conversation);if(terminal(job.status))break}await list(t);if(!valid(t))return;await refreshState(t);if(!valid(t))return}catch(error){showError(error,t)}finally{if(valid(t)&&submissionVersion===t.version){submissionVersion=null;controls()}}}
async function send(event){event?.preventDefault();if(!current||submissionVersion===viewVersion||changingVersion===viewVersion||state.active_request)return;const question=$('question').value,t=token();submissionVersion=t.version;controls();$('error').textContent='';try{const job=await api('/api/ask',{conversation_id:t.cid,question});if(!valid(t))return;$('question').value='';const conversation=await api(`/api/conversations/${t.cid}`);if(!valid(t))return;render(conversation);void poll(job.request_id,t)}catch(error){showError(error,t)}}
async function start(){const t=token();await refreshState(t);if(!valid(t))return;const conversations=await list(t);if(!valid(t)||conversations===null)return;let remembered=null;try{remembered=localStorage.getItem('day29-full-chat-conversation')}catch{}const found=conversations.find(c=>c.conversation_id===remembered)||conversations[0];if(found)await open(found.conversation_id);else await newChat()}
$('composer').addEventListener('submit',send);$('new').addEventListener('click',newChat);$('reset').addEventListener('click',reset);$('conversations').addEventListener('change',()=>open($('conversations').value));$('question').addEventListener('keydown',event=>{if(event.key==='Enter'&&(event.ctrlKey||event.metaKey)){event.preventDefault();void send()}});
setInterval(()=>{const t=token();void refreshState(t).catch(error=>showError(error,t))},1500);start().catch(error=>showError(error,token()));
