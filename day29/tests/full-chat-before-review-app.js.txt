const $=id=>document.getElementById(id);let current=null,viewVersion=0,state={active_request:null},sending=false,sendVersion=null;
const kinds={assignment:'Условие',discussion:'Обсуждение',personal_note:'Личная заметка'};
const terminal=s=>!['running','submitted'].includes(s);
const seconds=x=>typeof x==='number'?`${x.toFixed(2)} с`:'не измерено';
function node(tag,text,className){const e=document.createElement(tag);if(text!==undefined)e.textContent=text;if(className)e.className=className;return e}
async function api(path,body){const r=await fetch(path,body===undefined?{}:{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(body)});const value=await r.json();if(!r.ok)throw Error(value.error||`HTTP ${r.status}`);return value}
function controls(){const active=state.active_request;const busy=sending||!!active;$('send').disabled=busy||!current;$('status').textContent=active?(active.conversation_id===current?'Локальный запрос выполняется…':'Завершается запрос другого разговора'):sending?'Отправка…':'Готов к вопросу'}
async function refreshState(){state=await api('/api/state');$('scope').textContent=`Все ${state.index.chunk_count} фрагмента · ${state.index.document_count} документа · ${state.model}`;$('identity').textContent=`Scope: all\nИндекс: ${state.index.generation}\nSHA: ${state.index.db_sha256}\nTemperature ${state.options.temperature} · ctx ${state.options.num_ctx} · output ${state.options.num_predict} · think=false`;controls()}
function sourceView(source){const box=node('details',undefined,'source');box.dataset.requestId=source.request_id;box.append(node('summary',`${source.source_ref} · ${kinds[source.source]||source.source} · ${source.document_id} · ${source.chunk_id}`));for(const part of source.excerpts||[]){const p=node('div',undefined,'part');const author=part.attribution==='ambiguous'?'Атрибуция неоднозначна':part.author||'Автор неизвестен';p.append(node('div',`${author} · ${part.date||'Дата неизвестна'} · позиции ${part.start_char??'?'}–${part.end_char??'?'}`,'attribution'),node('pre',part.text));box.append(p)}const whole=node('details',undefined,'whole');whole.append(node('summary','Оригинальный фрагмент целиком'),node('pre',source.text,'source-full'));box.append(whole);return box}
function render(conversation){$('messages').replaceChildren();if(!conversation.turns.length){$('messages').append(node('p','Задайте новый вопрос. История этого разговора пока пуста.','empty'));return}for(const turn of conversation.turns){const box=node('article',undefined,'turn');box.dataset.requestId=turn.request_id;box.dataset.state=turn.status;if(turn.status==='error'||turn.status==='interrupted')box.classList.add('error');box.append(node('p',turn.question,'question'));let answer=turn.answer?.answer||'';if(!terminal(turn.status))answer='Ищу источники и выполняю локальную генерацию…';else if(turn.status==='interrupted')answer='Запрос был прерван перезапуском. Автоматического повтора не было.';else if(turn.error)answer=`${turn.error.phase}: ${turn.error.message}`;box.append(node('p',answer,'answer'));if(turn.query?.intent==='followup')box.append(node('p',`Продолжение темы: ${turn.query.anchor}`,'meta'));box.append(node('p',`request ${turn.request_id} · ${turn.status} · генерация ${seconds(turn.times?.generation)} · весь запрос ${seconds(turn.times?.total)}`,'meta'));if(turn.citations_valid)box.append(node('p','Ссылки проверены по свежему контексту. Смысловая поддержка автоматически не оценивалась.','support'));for(const source of turn.answer?.sources||[])box.append(sourceView(source));$('messages').append(box)}}
async function list(){const result=await api('/api/conversations');$('conversations').replaceChildren();for(const c of result.conversations){const option=node('option',`Разговор ${c.conversation_id.slice(0,6)} · ${c.turn_count} реплик`);option.value=c.conversation_id;$('conversations').append(option)}if(current)$('conversations').value=current;return result.conversations}
async function open(cid){viewVersion++;current=cid;sending=false;sendVersion=null;try{localStorage.setItem('day29-full-chat-conversation',cid)}catch{}const version=viewVersion;const c=await api(`/api/conversations/${cid}`);if(version!==viewVersion)return;render(c);await list();await refreshState();const pending=c.turns.find(t=>!terminal(t.status));if(pending)void poll(pending.request_id,cid,version)}
async function newChat(){const c=await api('/api/conversations',{});$('question').value='';$('error').textContent='';await open(c.conversation_id)}
async function reset(){if(!current)return;await api('/api/reset',{conversation_id:current});$('question').value='';$('error').textContent='';await open(current)}
async function poll(rid,cid,version){try{for(;;){await new Promise(r=>setTimeout(r,750));if(current!==cid||viewVersion!==version)return;const job=await api(`/api/requests/${rid}`);if(current!==cid||viewVersion!==version)return;render(await api(`/api/conversations/${cid}`));if(terminal(job.status))break}await list();await refreshState()}catch(e){if(current===cid&&viewVersion===version)$('error').textContent=String(e)}finally{if(sendVersion===version){sending=false;sendVersion=null;controls()}}}
async function send(event){event?.preventDefault();if(!current||sending||state.active_request)return;const q=$('question').value;const cid=current,version=viewVersion;sending=true;sendVersion=version;controls();$('error').textContent='';try{const job=await api('/api/ask',{conversation_id:cid,question:q});if(current!==cid||viewVersion!==version){if(sendVersion===version){sending=false;sendVersion=null;controls()}return}$('question').value='';render(await api(`/api/conversations/${cid}`));void poll(job.request_id,cid,version)}catch(e){if(sendVersion===version){$('error').textContent=String(e);sending=false;sendVersion=null;controls()}}}
async function start(){await refreshState();const conversations=await list();let remembered=null;try{remembered=localStorage.getItem('day29-full-chat-conversation')}catch{}const found=conversations.find(c=>c.conversation_id===remembered)||conversations[0];if(found)await open(found.conversation_id);else await newChat();$('composer').addEventListener('submit',send);$('new').addEventListener('click',()=>newChat().catch(e=>$('error').textContent=String(e)));$('reset').addEventListener('click',()=>reset().catch(e=>$('error').textContent=String(e)));$('conversations').addEventListener('change',()=>open($('conversations').value).catch(e=>$('error').textContent=String(e)));$('question').addEventListener('keydown',e=>{if(e.key==='Enter'&&(e.ctrlKey||e.metaKey)){e.preventDefault();void send()}});setInterval(()=>refreshState().catch(()=>{}),1500)}start().catch(e=>$('error').textContent=String(e));
