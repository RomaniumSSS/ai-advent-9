def file_sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def code_hashes():
    root=Path(__file__).resolve().parent
    return {name:file_sha(root/name) for name in ('quality_eval.py','full_chat_core.py','core.py','metrics.py')}


def protocol():
    return {'version':VERSION,'model':core.MODEL,'digest':core.GEN_DIGEST,'think':False,
            'baseline_profile':chat.PROFILE,'critic_system':CRITIC_SYSTEM,'repair_system':REPAIR_SYSTEM,
            'critic_schema':CRITIC_SCHEMA,'repair_schema':core.SOURCE_SCHEMA,
            'critic_options':{**chat.OPTIONS,'num_predict':512},'repair_options':chat.OPTIONS,
            'max_repairs_per_pair':1,'transport_timeout':core.TIMEOUT,'walltime_limit':None,'call_cap':None}


def private_dir(path):
    path=Path(path).absolute()
    if any(item.is_symlink() for item in (path,*path.parents)):raise ValueError('symlink_private_root')
    path=path.resolve()
    missing=[];cursor=path
    while not cursor.exists():missing.append(cursor);cursor=cursor.parent
    for item in reversed(missing):item.mkdir(mode=0o700)
    path.chmod(0o700)
    return path


@contextmanager
def controller_lock(root):
    # Глобальный model lock защищает Ollama; этот отдельный lock защищает
    # один checkpoint от двух CLI controllers, включая recovery до dispatch.
    fd=os.open(Path(root)/'runner.lock',os.O_WRONLY|os.O_CREAT,0o600)
    try:
        os.fchmod(fd,0o600)
        try:fcntl.flock(fd,fcntl.LOCK_EX|fcntl.LOCK_NB)
        except BlockingIOError:raise ValueError('run_controller_busy')
        yield
    finally:os.close(fd)


def validate_matrix(matrix):
    cases=matrix.get('cases');order=matrix.get('order')
    if not isinstance(cases,list) or not cases:raise ValueError('empty_cases')
    ids=[c.get('id') for c in cases]
    if any(not isinstance(cid,str) or not re.fullmatch(r'[A-Z][0-9]{2}',cid) for cid in ids) or len(set(ids))!=len(ids):raise ValueError('invalid_case_ids')
    if not isinstance(order,list) or set(order)!=set(ids) or len(order)!=len(ids):raise ValueError('invalid_order')
    by_id={c['id']:c for c in cases};groups={}
    for c in cases:
        core.validate_question(c['question'])
        if c.get('split') not in ('pilot','holdout') or not isinstance(c.get('semantic_group'),str):raise ValueError('invalid_split_group')
        if c.get('primary_type') not in CASE_TYPES:raise ValueError('invalid_case_type')
        if type(c.get('new_conversation')) is not bool:raise ValueError('missing_conversation_policy')
        group=c['semantic_group'];split=c['split']
        if group in groups and groups[group]!=split:raise ValueError('group_split_leak')
        groups[group]=split;dep=c.get('dependency')
        if dep is not None:
            if dep not in by_id or order.index(dep)>=order.index(c['id']):raise ValueError('dependency_not_before_case')
            if by_id[dep]['split']!=split or by_id[dep]['semantic_group']!=group or c['new_conversation']:raise ValueError('invalid_dependency_chain')
    return by_id


def selection_for(matrix,cases_sha,case_ids=None,selection_file=None):
    if case_ids is not None and selection_file is not None:raise ValueError('selection_sources_conflict')
    supplied=json.loads(Path(selection_file).read_text()) if selection_file else {}
    if selection_file and (not isinstance(supplied,dict) or supplied.get('cases_file_sha256')!=cases_sha or 'case_ids' not in supplied):raise ValueError('selection_dataset_mismatch')
    requested=case_ids if case_ids is not None else supplied.get('case_ids',matrix['order'])
    if (not isinstance(requested,list) or not requested or any(not isinstance(cid,str) for cid in requested) or
        len(set(requested))!=len(requested) or not set(requested)<=set(matrix['order'])):raise ValueError('invalid_selected_case_ids')
    by_id={c['id']:c for c in matrix['cases']};selected=set(requested)
    for cid in requested:
        if by_id[cid].get('dependency') and by_id[cid]['dependency'] not in selected:raise ValueError('selection_dependency_missing')
    return {'case_ids':[cid for cid in matrix['order'] if cid in selected],'cases_file_sha256':cases_sha,
            'source_sha256':file_sha(selection_file) if selection_file else None,
            'seed':supplied.get('seed'),'method':supplied.get('method','explicit_ids' if case_ids is not None else 'all_frozen_cases')}


def prepare(cases_file,output_root,audit_file=None,case_ids=None,selection_file=None):
    cases_file=Path(cases_file);matrix=json.loads(cases_file.read_text());validate_matrix(matrix)
    audit=json.loads(Path(audit_file).read_text()) if audit_file else None
    sha=file_sha(cases_file)
    selection=selection_for(matrix,sha,case_ids,selection_file)
    if matrix.get('profile') is not None and matrix['profile']!=chat.PROFILE:raise ValueError('baseline_profile_mismatch')
    for name,digest in matrix.get('baseline_hashes',{}).items():
        if name not in ('core.py','full_chat.py','full_chat_core.py') or file_sha(Path(__file__).parent/name)!=digest:raise ValueError('baseline_code_mismatch')
    if audit is not None:validate_audit(audit,sha,matrix)
    root=private_dir(output_root)/uuid.uuid4().hex;private_dir(root)
    manifest={'run_id':root.name,'prepared_at':chat.now(),'mode':'prepare_only','cases_file_sha256':sha,
              'inputs_sha256':core.sha(matrix),'protocol':protocol(),'protocol_sha256':core.sha(protocol()),
              'code_hashes':code_hashes(),'index':matrix.get('index'),'profile':chat.PROFILE,
              'audit':audit,'audit_kind':'agent_source_audit_not_human_acceptance','order':matrix['order'],
              'selection':selection,'selection_sha256':core.sha(selection)}
    core.save_private(root/'inputs.json',matrix);core.save_private(root/'manifest.json',manifest)
    core.save_private(root/'selection.json',selection)
    core.save_private(root/'checkpoint.json',{'run_id':root.name,'cases':{},'paused':False,'idle_confirmations':[],'events':[]})
    return root


def validate_audit(audit,cases_file_sha256,matrix):
    if (audit.get('approved') is not True or audit.get('cases_file_sha256')!=cases_file_sha256 or
        audit.get('audit_kind')!='independent_agent_source_audit' or audit.get('human_acceptance') is not False):raise ValueError('gold_audit_mismatch')
    for key in ('cases_gold_sha256','evidence_sha256'):
        if matrix.get(key) is not None and audit.get(key)!=matrix[key]:raise ValueError('gold_audit_mismatch')


def critique_payload(receipt,answer,issues=None,stage='critic'):
    if stage not in ('critic','repair','recheck'):raise ValueError('invalid_stage')
    # AICODE-NOTE: берём фактически переданный A body, не повторяем retrieval и
    # не читаем gold; ни history, ни sources, ни draft для B не усекаются.
    if receipt['messages'][0]!={'role':'system','content':chat.SYSTEM}:raise ValueError('baseline_system_mismatch')
    original=json.loads(receipt['messages'][1]['content'])
    keys=('current_question','resolved_query','history_for_dialogue_only','sources')
    if set(original)!=set(keys):raise ValueError('baseline_context_shape')
    body={key:copy.deepcopy(original[key]) for key in keys}
    body['draft']={'answer':answer['answer'],'source_refs':answer['source_refs']}
    if stage=='repair':body['issues']=copy.deepcopy(issues)
    schema=core.SOURCE_SCHEMA if stage=='repair' else CRITIC_SCHEMA
    system=REPAIR_SYSTEM if stage=='repair' else CRITIC_SYSTEM
    options={**chat.OPTIONS,'num_predict':1024 if stage=='repair' else 512}
    payload={'model':core.MODEL,'stream':False,'think':False,'format':copy.deepcopy(schema),'options':options,
             'keep_alive':'5m','messages':[{'role':'system','content':system},{'role':'user','content':core.packed(body).decode()}]}
    check_payload_budget(payload)
    return payload


def check_payload_budget(payload):
    # Весь сериализованный envelope включает schema/options/flags, плюс запас
    # шаблона и выход. Это консервативная byte bound, а не измерение tokenizer.
    count=len(core.packed(payload))+core.TEMPLATE_RESERVE;options=payload['options']
    if count+options['num_predict']>options['num_ctx']:raise ValueError('stage_budget_rejected')
    return {'payload_bytes':len(core.packed(payload)),'input_upper_bound':count,
            'output_reserved':options['num_predict'],'num_ctx':options['num_ctx']}


def validate_critique(value,refs):
    if not isinstance(value,dict) or set(value)!= {'verdict','issues'}:raise ValueError('invalid_critic_shape')
    verdict=value['verdict'];issues=value['issues']
    if verdict not in ('accept','repair','insufficient_evidence') or not isinstance(issues,list) or len(issues)>4:raise ValueError('invalid_verdict_or_issues')
    if (verdict=='accept' and issues) or (verdict=='repair' and not issues):raise ValueError('verdict_issues_mismatch')
    for item in issues:
        if not isinstance(item,dict) or set(item)!= {'type','claim','reason','source_refs'}:raise ValueError('invalid_issue_shape')
        if item['type'] not in ISSUE_TYPES or any(not isinstance(item[k],str) or not item[k].strip() for k in ('claim','reason')):raise ValueError('invalid_issue_text')
        links=item['source_refs']
        if not isinstance(links,list) or any(not isinstance(ref,str) or ref not in refs for ref in links) or len(set(links))!=len(links) or len(links)>5:raise ValueError('invalid_issue_refs')
        if not links and verdict!='insufficient_evidence':raise ValueError('issue_without_evidence')
    return value


def validate_response(raw,payload,refs):
    if not isinstance(raw,dict) or raw.get('model')!=core.MODEL or raw.get('done') is not True or raw.get('done_reason')!='stop':raise ValueError('incomplete_or_wrong_model_response')
    count=raw.get('prompt_eval_count')
    if type(count) is not int or count<=0:raise ValueError('missing_prompt_count')
    if count+payload['options']['num_predict']>payload['options']['num_ctx']:raise ValueError('observed_context_overflow')
    def unique_keys(pairs):
        result={}
        for key,value in pairs:
            if key in result:raise ValueError('duplicate_json_key')
            result[key]=value
        return result
    value=json.loads(raw['message']['content'],object_pairs_hook=unique_keys)
    if payload['format']==CRITIC_SCHEMA:return validate_critique(value,refs)
    if not isinstance(value,dict) or set(value)!= {'answer','source_refs'} or not isinstance(value['answer'],str) or not value['answer'].strip():raise ValueError('invalid_repair_shape')
    links=value['source_refs']
    if not isinstance(links,list) or any(not isinstance(ref,str) or ref not in refs for ref in links) or len(set(links))!=len(links) or len(links)>5:raise ValueError('invalid_repair_refs')
    if not links:
        # Тот же unknown guard, что у A; мнение recheck не разрешает
        # фактический ответ без refs. Смысл unknown всё ещё оценивает assessor.
        core.validate_answer({'answer':value['answer'],'sources':[],'quotes':[]},[])
    return value


def run_b(receipt,dispatch):
    if receipt.get('status') not in ('done','unknown') or not receipt.get('citations_valid'):
        return {'status':'dependency_blocked','quality':'NOT_CHECKED'}
    if not receipt.get('generation_called'):
        return {'status':'not_applicable','quality':'NOT_CHECKED','answer':receipt.get('answer')}
    a=receipt['answer'];refs={s['source_ref'] for s in json.loads(receipt['messages'][1]['content'])['sources']}
    candidate=None;stage='critic'
    try:
        critique=dispatch(stage,critique_payload(receipt,a),refs)
        if critique['verdict']=='accept':return {'status':'accepted_by_critic','answer':a,'quality':'NOT_CHECKED','critic':critique}
        if critique['verdict']=='insufficient_evidence':return {'status':'unavailable','quality':'NOT_CHECKED','critic':critique,'reason':'insufficient_evidence'}
        stage='repair';candidate=dispatch(stage,critique_payload(receipt,a,critique['issues'],stage),refs)
        stage='recheck';recheck=dispatch(stage,critique_payload(receipt,candidate,stage=stage),refs)
        return {'status':'accepted_by_recheck' if recheck['verdict']=='accept' else 'unverified',
                'answer':candidate,'quality':'NOT_CHECKED','critic':critique,'recheck':recheck}
    except Exception as error:
        return {'status':'unverified' if stage=='recheck' else 'unavailable','quality':'NOT_CHECKED',
                'candidate':candidate,'failed_stage':stage,'reason':str(error)[:300],'error_type':type(error).__name__}


def execute_pair(engine,case,turns,client):
    """Малый in-memory контракт для тестов; durable execution делает Runner."""
    a=engine.ask(case['question'],turns,'synthetic');calls=int(a.get('generation_called',False));seconds=0
    def dispatch(stage,payload,refs):
        nonlocal calls,seconds
        check_payload_budget(payload);calls+=1;start=time.monotonic()
        try:return validate_response(client.request('/api/chat',payload),payload,refs)
        finally:seconds+=time.monotonic()-start
    b=run_b(a,dispatch)
    return {'A':a,'B':b,'generation_calls':calls,'added_seconds':seconds,
            'A_latency':a.get('times',{}).get('total'),'B_latency':(a.get('times',{}).get('total') or 0)+seconds}


class CheckpointClient:
    """Запись каждого HTTP dispatch раньше transport; ни retry, ни busy guesses."""
    def __init__(self,base,runner):self.base=base;self.runner=runner;self.network=[]
    def verify(self,profile=None):return core.Ollama.verify(self,profile)
    def embed(self,question):return core.Ollama.embed(self,question)
    def request(self,path,payload=None):
        if path=='/api/chat':check_payload_budget(payload)
        event={'case_id':self.runner.case_id,'stage':self.runner.stage,'path':path,
               'payload':copy.deepcopy(payload),'status':'dispatching','started_at':chat.now(),'started_epoch':time.time()}
        self.runner.state['events'].append(event);self.runner.save()
        print(json.dumps({'case_id':self.runner.case_id,'stage':self.runner.stage,'status':'dispatching',
                          'call_type':'generation' if path=='/api/chat' else 'embedding' if path=='/api/embed' else 'service',
                          'counters':self.runner.counters()}),flush=True)
        self.network.append({'url':core.OLLAMA_URL+path,'method':'POST' if payload is not None else 'GET'})
        start=time.monotonic()
        try:
            raw=self.base.request(path,payload);event.update(status='done',response=raw)
            if path=='/api/chat' and (not isinstance(raw,dict) or raw.get('done') is not True):
                event['status']='interrupted_unknown'
                self.runner.state.update(paused=True,pause_started_epoch=time.time(),pause_reason='provider_completion_unknown')
            return raw
        except Exception as error:
            uncertain=path in ('/api/chat','/api/embed')
            event.update(status='interrupted_unknown' if uncertain else 'error',error_type=type(error).__name__,error=str(error)[:300])
            if uncertain:self.runner.state.update(paused=True,pause_started_epoch=time.time(),pause_reason='transport_execution_unknown')
            raise
        finally:
            event.update(seconds=time.monotonic()-start,finished_at=chat.now());self.runner.save()


class Runner:
    def __init__(self,root,engine_factory,base_client,lock_factory=core.generation_lock,live=False):
        self.root=private_dir(root);self.manifest=json.loads((self.root/'manifest.json').read_text())
        self.inputs=json.loads((self.root/'inputs.json').read_text());self.cases=validate_matrix(self.inputs)
        self.selection=json.loads((self.root/'selection.json').read_text())
        if self.selection!=self.manifest.get('selection') or core.sha(self.selection)!=self.manifest.get('selection_sha256'):
            raise ValueError('immutable_selection_hash_mismatch')
        selected=self.selection.get('case_ids',[])
        validated=selection_for(self.inputs,self.manifest['cases_file_sha256'],case_ids=selected)
        if selected!=validated['case_ids'] or self.selection['cases_file_sha256']!=self.manifest['cases_file_sha256']:raise ValueError('invalid_frozen_selection')
        self.lock_factory=lock_factory;self.case_id=None;self.stage=None
        if (self.manifest['inputs_sha256']!=core.sha(self.inputs) or self.manifest['protocol_sha256']!=core.sha(protocol()) or
            self.manifest['protocol']!=protocol() or self.manifest['order']!=self.inputs['order'] or
            self.manifest['profile']!=chat.PROFILE or self.manifest['code_hashes']!=code_hashes()):raise ValueError('immutable_run_hash_mismatch')
        if live:
            if not self.manifest.get('audit'):raise ValueError('agent_source_audit_required')
            validate_audit(self.manifest['audit'],self.manifest['cases_file_sha256'],self.inputs)
        self.client=CheckpointClient(base_client,self);self.engine=engine_factory(self.client)
        with controller_lock(self.root):self.reload_recover()

    def reload_recover(self):
        self.state=json.loads((self.root/'checkpoint.json').read_text())
        recovered=False
        for case in self.state['cases'].values():
            for stage in case.get('stages',{}).values():
                if stage.get('status') in ('running','dispatching'):
                    stage.update(status='interrupted_unknown',recovered_at=chat.now());recovered=True
        for event in self.state['events']:
            if event['status']=='dispatching':event.update(status='interrupted_unknown',recovered_at=chat.now());recovered=True
        if recovered:
            self.state.update(paused=True,pause_started_epoch=self.state.get('pause_started_epoch',time.time()),pause_reason='recovered_ambiguous_inflight');self.save()

    def save(self):core.save_private(self.root/'checkpoint.json',self.state)

    def confirm_idle(self,note):
        if not isinstance(note,str) or not note.strip():raise ValueError('operator_idle_note_required')
        with controller_lock(self.root):
            self.reload_recover()
            self.state['idle_confirmations'].append({'at':chat.now(),'note':note,'kind':'explicit_operator_confirmation',
              'unconfirmed_seconds':max(0,time.time()-self.state.get('pause_started_epoch',time.time()))})
            self.state['paused']=False;self.save()

    def prefix(self,cid):
        chain=[];dep=self.cases[cid].get('dependency')
        while dep:
            previous=self.state['cases'].get(dep,{}).get('stages',{}).get('A',{})
            a=previous.get('result')
            if previous.get('status')!='done' or not a or a.get('status') not in ('done','unknown') or not a.get('answer'):raise ValueError('dependency_blocked')
            chain.append(copy.deepcopy(a));dep=self.cases[dep].get('dependency')
        return list(reversed(chain))

    def dispatch(self,stage,payload,refs):
        case=self.state['cases'][self.case_id];saved=case['stages'].get(stage)
        if saved:
            if saved.get('payload') is not None and saved['payload']!=payload:raise ValueError('stage_payload_immutability_violation')
            if saved['status']=='done':return saved['validated']
            raise ValueError(saved['status'])
        check_payload_budget(payload)
        with self.lock_factory():
            self.stage=stage;record={'status':'running','payload':payload,'budget':check_payload_budget(payload),'started_at':chat.now()}
            case['stages'][stage]=record;self.save();start=time.monotonic()
            try:
                self.client.verify(chat.PROFILE)
                raw=self.client.request('/api/chat',payload);record['raw']=raw;record['metrics']=ollama_metrics(raw)
                record['validated']=validate_response(raw,payload,refs);record['status']='done'
                return record['validated']
            except Exception as error:
                record.update(status='interrupted_unknown' if self.state['paused'] else 'error',error_type=type(error).__name__,reason=str(error)[:300]);raise
            finally:
                record.update(seconds=time.monotonic()-start,finished_at=chat.now());self.save()

    def run(self,split=None):
        with controller_lock(self.root):
            self.reload_recover()
            return self._run(split)

    def _run(self,split=None):
        if self.state['paused']:raise ValueError('operator_idle_confirmation_required_no_replay')
        for cid in self.manifest['order']:
            if cid not in self.selection['case_ids']:continue
            if split and self.cases[cid]['split']!=split:continue
            self.case_id=cid;self.stage='A';case=self.state['cases'].setdefault(cid,{'stages':{}})
            if case.get('status')=='complete':continue
            case_start=time.monotonic()
            try:turns=self.prefix(cid)
            except ValueError:
                case.update(status='dependency_blocked',B={'status':'dependency_blocked','quality':'NOT_CHECKED'});self.save();continue
            a_stage=case['stages'].get('A')
            if a_stage is None:
                case['prefix']=turns;case['prefix_sha256']=core.sha(turns)
                a_stage={'status':'running','started_at':chat.now()};case['stages']['A']=a_stage;self.save();start=time.monotonic()
                try:
                    a=self.engine.ask(self.cases[cid]['question'],turns,conversation_id=uuid.uuid4().hex,run_id=self.manifest['run_id'])
                    a_stage.update(result=a,status='interrupted_unknown' if self.state['paused'] else 'done')
                except Exception as error:a_stage.update(status='interrupted_unknown' if self.state['paused'] else 'error',error_type=type(error).__name__,reason=str(error)[:300])
                finally:a_stage.update(seconds=time.monotonic()-start,finished_at=chat.now());self.save()
            if a_stage['status']=='done':
                b_start=time.monotonic();case['B']=run_b(a_stage['result'],self.dispatch);case['status']='complete'
                case['b_execution_seconds']=case.get('b_execution_seconds',0)+time.monotonic()-b_start
            else:case.update(status='complete',B={'status':'dependency_blocked','quality':'NOT_CHECKED','reason':a_stage['status']})
            b=case['B']
            if b.get('failed_stage') and b['failed_stage'] not in case['stages']:
                case['stages'][b['failed_stage']]={'status':'budget_rejected' if b.get('reason')=='stage_budget_rejected' else 'error',
                    'reason':b.get('reason'),'error_type':b.get('error_type'),'seconds':0,'dispatch_count':0}
            a_error=a_stage.get('result',{}).get('error',{})
            wrong_response_model=any(e.get('response',{}).get('model')!=core.MODEL for e in self.state['events']
                                     if e['case_id']==cid and e['path']=='/api/chat' and isinstance(e.get('response'),dict))
            if a_error.get('phase')=='model_identity' or wrong_response_model or any(s.get('reason','').startswith(('model_digest_mismatch','remote_model_forbidden','missing_local_model_info','model_context_too_small')) for s in case['stages'].values()):
                self.state.update(paused=True,pause_reason='model_identity_failure',pause_started_epoch=time.time())
            case['execution_seconds']=case.get('execution_seconds',0)+time.monotonic()-case_start
            self.save();print(json.dumps({'case_id':cid,'stage':self.stage,'status':case['B']['status'],'counters':self.counters()},ensure_ascii=False),flush=True)
            if self.state['paused']:break
        summary=self.summary();core.save_private(self.root/'summary.json',summary);return summary

    def counters(self):
        events=self.state['events'];chats=[e for e in events if e['path']=='/api/chat'];embeds=[e for e in events if e['path']=='/api/embed']
        def response(e):return e['response'] if isinstance(e.get('response'),dict) else {}
        def observed(key):return [response(e)[key] for e in chats if type(response(e).get(key)) is int and response(e)[key]>=0]
        prompt=observed('prompt_eval_count');output=observed('eval_count')
        return {'generation_calls':len(chats),'embedding_calls':len(embeds),'service_calls':len(events)-len(chats)-len(embeds),
                'http_seconds':sum(e.get('seconds',0) for e in events),
                'prompt_tokens_observed':sum(prompt),'output_tokens_observed':sum(output),
                'prompt_tokens_total':sum(prompt) if len(prompt)==len(chats) else None,
                'output_tokens_total':sum(output) if len(output)==len(chats) else None,
                'missing_generation_metrics':sum(any(type(response(e).get(k)) is not int for k in ('eval_count','prompt_eval_count')) for e in chats),
                'ambiguous_http_calls':sum(e['status']=='interrupted_unknown' for e in events),
                'http_errors':sum(e['status']=='error' for e in events)}

    def summary(self):
        rows=[]
        for cid in self.manifest['order']:
            case=self.state['cases'].get(cid,{});a=case.get('stages',{}).get('A',{});b=case.get('B',{})
            seconds=sum(s.get('seconds',0) for name,s in case.get('stages',{}).items() if name!='A')
            a_seconds=a.get('result',{}).get('times',{}).get('total')
            rows.append({'case_id':cid,'split':self.cases[cid]['split'],'type':self.cases[cid]['primary_type'],
                         'group_sha256':core.sha(self.cases[cid]['semantic_group']),
                         'selected':cid in self.selection['case_ids'],
                         'status':case.get('status','pending') if cid in self.selection['case_ids'] else 'pending_unselected',
                         'B_status':b.get('status','pending') if cid in self.selection['case_ids'] else 'pending_unselected',
                         'quality':'NOT_CHECKED','A_latency':a_seconds,'added_observed_seconds':case.get('b_execution_seconds'),
                         'added_stage_seconds':seconds,'pair_observed_seconds':case.get('execution_seconds'),
                         'B_latency':a_seconds+case.get('b_execution_seconds',seconds) if a_seconds is not None and b.get('status') in ('accepted_by_critic','accepted_by_recheck') else None})
        holdout=[r for r in rows if r['split']=='holdout'];selected_holdout=[r for r in holdout if r['selected']]
        return {'run_id':self.manifest['run_id'],'protocol_sha256':self.manifest['protocol_sha256'],
                'selection_sha256':self.manifest['selection_sha256'],'selected_case_ids':self.selection['case_ids'],
                'selected_case_count':len(self.selection['case_ids']),'holdout_total':len(holdout),
                'selected_holdout_count':len(selected_holdout),'pending_unselected_holdout':len(holdout)-len(selected_holdout),
                'holdout_claim':'partial_selected_subset' if len(selected_holdout)<len(holdout) else 'whole_selected_holdout_not_yet_quality_scored',
                'counters':self.counters(),
                'cases':rows,'paused':self.state['paused'],'semantic_assessment':'NOT_CHECKED',
                'completed_pairs':sum(r['B_status'] in ('accepted_by_critic','accepted_by_recheck') for r in rows),
                'idle_wait_seconds':sum(c['unconfirmed_seconds'] for c in self.state['idle_confirmations']),
                'paid_provider_calls':0,'human_acceptance':False}

def main(argv=None):
    parser=argparse.ArgumentParser(description=__doc__);parser.add_argument('command',nargs='?',choices=('prepare','mock-dry-run','run','summary'),default='prepare')
    parser.add_argument('--cases',type=Path,default=Path(__file__).resolve().parent/'private/full-chat/quality-eval/cases-v1.json')
    parser.add_argument('--output-root',type=Path,default=Path(__file__).resolve().parent/'private/full-chat/quality-eval/runs')
    parser.add_argument('--audit-file',type=Path);parser.add_argument('--run-root',type=Path);parser.add_argument('--index-root',type=Path)
    parser.add_argument('--case-id',action='append',help='prepare: frozen subset; run: exact selection guard')
    parser.add_argument('--selection-file',type=Path,help='prepare: root-owned seeded selection manifest')
    parser.add_argument('--split',choices=('pilot','holdout'));parser.add_argument('--confirm-idle',help='Записанное подтверждение оператора; /api/ps не является busy proof')
    args=parser.parse_args(argv)
    allowed=(Path(__file__).resolve().parent/'private/full-chat/quality-eval').resolve()
    for path in (args.output_root,args.run_root):
        if path is not None and not path.resolve().is_relative_to(allowed):parser.error('CLI private paths must remain under private/full-chat/quality-eval')
    if args.command=='prepare':
        if args.audit_file is None:
            default_audit=allowed/'source-audit-v1.json'
            if default_audit.exists():args.audit_file=default_audit
        root=prepare(args.cases,args.output_root,args.audit_file,args.case_id,args.selection_file)
        selection=json.loads((root/'selection.json').read_text())
        print(json.dumps({'run_id':root.name,'mode':'prepare_only','network_calls':0,'selected_case_ids':selection['case_ids'],'selection_sha256':core.sha(selection)}));return 0
    if args.command=='mock-dry-run':
        matrix=mock_matrix();path=private_dir(args.output_root)/('mock-input-'+uuid.uuid4().hex+'.json');core.save_private(path,matrix)
        root=prepare(path,args.output_root);chunk={'chunk_id':'invented','document_id':'mock','source':'discussion','text':'Макет хранится на полке.',
             'vector':[1.0]+[0.0]*1023,'start_char':0,'end_char':len('Макет хранится на полке.')}

                  'completed_pairs':summary['completed_pairs'],'not_applicable':sum(c['B_status']=='not_applicable' for c in summary['cases']),

    manifest=json.loads((args.run_root/'manifest.json').read_text())
    if args.selection_file is not None:parser.error('--selection-file freezes at prepare, not run')
    if args.case_id is not None and (len(args.case_id)!=len(set(args.case_id)) or set(args.case_id)!=set(manifest['selection']['case_ids'])):
        parser.error('--case-id must exactly match frozen selection')

    summary=runner.run(args.split);return 2 if summary['paused'] else 0
